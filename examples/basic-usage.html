基础使用方法

// 1. 最简单的使用方式 const canvas = document.getElementById('revix-canvas'); const revix = createExample(canvas); // 2. 从 JSON 数据创建 const jsonData = { bounds: [-100, 100, 100, -100], backgroundColor: '#ff9999', children: [...] }; const revix = createFromJSON(canvas, jsonData); // 3. 自定义配置 const revix = createRevix(canvas, { viewTree: customViewTree, enableInspector: true, autoRender: true }); // 4. 监听事件 revix.on('nodeSelected', (node) => { console.log('选中节点:', node); });
